funciones escalares.- Devuelve un solo valor basado en los argumentos que se le pasa
crear una funcion
--crea una funcion para calcular el total de ventas para un cliente en especifico
--para modificar la funcion en vez de create ponemos alter
create function sales.calcularTotalVentasPorCliente(@customerID int) --los parametros van con arroba 
returns decimal(18,2) --parametros de decimal, hasta 18 digitos y dos decimales, Tambien podemos poner float, es lo que esperamos de la funcion
as 
begin --dentro de begin esta la logica de la funcion
declare @totalVentas decimal(18,2) --declaramos una variable
set @totalVentas=(select sum(soh.totalDue)
from sales.salesOrderHeader soh
inner join sales.customer c on soh.customerID = c.customerID
where c.customerID=@customerID)
return @totalVentas;
end;

en otro query llamamos a la funcion
select customerID,
sales.calcularTotalVentasPorCliente(customerID)
from sales.customer
